# obserae demo — file manifest fetched by install.sh (one relative path per line).
# Single source of truth for the file set, so adding a demo file only needs a
# push here — install.sh (deployed separately on demo.obserae.com) stays untouched.
# Blank lines and lines starting with '#' are ignored.
#
# Regenerate from the demo directory with:
#   find . -type f -not -path './.git/*' -not -name install.sh \
#     -not -name manifest.txt -not -path './freeipa/ca.crt' -not -name '.env' \
#     | sed 's|^\./||' | sort
#
# Excluded on purpose: install.sh (the bootstrap itself), manifest.txt (this file),
# freeipa/ca.crt (regenerated by scripts/setup-ldap.sh), .env (host-local overrides).
